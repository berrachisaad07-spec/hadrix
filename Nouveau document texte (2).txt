def normalize_time(time_str):
    """تحويل الوقت إلى صيغة 24 ساعة"""
    time_str = time_str.strip()
    if 'مساءً' in time_str or 'م' in time_str:
        time_str = re.sub(r'[مساءًم]', '', time_str).strip()
        h, m = time_str.split(':')
        h = 12 if h == 12 else (int(h) % 12) + 12
        return f"{h:02d}:{int(m):02d}"
    elif 'صباحًا' in time_str or 'ص' in time_str:
        time_str = re.sub(r'[صباحًاص]', '', time_str).strip()
        h, m = time_str.split(':')
        return f"{int(h):02d}:{int(m):02d}"
    else:
        parts = time_str.split(':')
        return f"{int(parts[0]):02d}:{int(parts[1]):02d}"

def parse_matches(text):
    """تحليل قائمة المباريات وتجميع الروابط"""
    matches_dict = {}
    for line in text.split('\n'):
        line = line.strip()
        if not line:
            continue
        match = re.match(r'(\d{1,2}:\d{2}(?:\s*(?:صباحًا|مساءً|ص|م))?)\s+(.+?)\s*[×xXضدvs]+\s*(.+?)\s*\|\s*(https?://.+)', line, re.IGNORECASE)
        if match:
            time_str = match.group(1)
            team1 = match.group(2).strip()
            team2 = match.group(3).strip()
            url = match.group(4).strip()
            
            time_24 = normalize_time(time_str)
            is_special = any(x in team1.lower() for x in ['rugby', 'nba'])
            
            if is_special:
                key = f"{team1}||{time_24}"
            else:
                key = f"{team1}||{team2}||{time_24}"
            
            if key not in matches_dict:
                matches_dict[key] = {
                    'time': time_24,
                    'team1': team1,
                    'team2': '' if is_special else team2,
                    'urls': [],
                    'is_special': is_special
                }
            
            if url not in matches_dict[key]['urls']:
                matches_dict[key]['urls'].append(url)
    
    return list(matches_dict.values())

# 9. ⭐ Blogger API - لنشر Pages (صفحات) ⭐
# ==========================================
class BloggerManager:
    """مدير Blogger API لنشر صفحات (Pages) - يدعم مدونات متعددة"""
    
    def __init__(self, config):
        self.config = config
        self.service = None
        self.last_error = None
        self.debug_log = []
    
    def log(self, message, level="INFO"):
        """تسجيل رسالة في السجل"""
        timestamp = datetime.now().strftime('%H:%M:%S')
        log_entry = f"[{timestamp}] [{level}] {message}"
        self.debug_log.append(log_entry)
        if len(self.debug_log) > 100:
            self.debug_log.pop(0)
        print(log_entry)
    
    def check_prerequisites(self):
        """فحص المتطلبات الأساسية"""
        self.log(f"🔍 فحص المتطلبات لـ {self.config['name']}...")
        issues = []
        
        cred_file = self.config['credentials_file']
        if not os.path.exists(cred_file):
            issues.append(f"❌ ملف {cred_file} غير موجود!")
            self.log(f"❌ الملف مفقود: {cred_file}", "ERROR")
        else:
            try:
                with open(cred_file, 'r', encoding='utf-8') as f:
                    data = json.load(f)
                    if 'installed' not in data and 'web' not in data:
                        issues.append("❌ ملف credentials.json غير صالح!")
                    else:
                        self.log(f"✅ ملف {cred_file} موجود وصالح")
            except Exception as e:
                issues.append(f"❌ خطأ في قراءة credentials.json: {e}")
        
        required_libs = ['googleapiclient', 'google_auth_oauthlib', 'google.auth']
        for lib in required_libs:
            try:
                __import__(lib.split('.')[0])
                self.log(f"✅ المكتبة {lib} مثبتة")
            except ImportError:
                issues.append(f"❌ المكتبة {lib} غير مثبتة!")
        
        blog_id = self.config['blog_id']
        if not blog_id or len(blog_id) < 10:
            issues.append("❌ Blog ID غير صحيح!")
        else:
            self.log(f"✅ Blog ID: {blog_id}")
        
        return issues
    
    def authenticate(self, force_new=False):
        """المصادقة مع Blogger API"""
        try:
            from googleapiclient.discovery import build
            from google_auth_oauthlib.flow import InstalledAppFlow
            from google.auth.transport.requests import Request
            from google.oauth2.credentials import Credentials
            
            self.log(f"🔐 بدء المصادقة لـ {self.config['name']}...")
            
            SCOPES = ['https://www.googleapis.com/auth/blogger']
            creds = None
            
            if force_new and os.path.exists(self.config['token_file']):
                os.remove(self.config['token_file'])
                self.log("🗑️ تم حذف token القديم")
            
            token_file = self.config['token_file']
            if os.path.exists(token_file):
                try:
                    self.log(f"📂 جاري تحميل token من: {token_file}")
                    creds = Credentials.from_authorized_user_file(token_file, SCOPES)
                    self.log("✅ تم تحميل token بنجاح")
                except Exception as e:
                    self.log(f"⚠️ خطأ في تحميل token: {e}", "WARNING")
                    creds = None
            
            if creds and creds.valid:
                self.log("✅ Token صالح")
            elif creds and creds.expired and creds.refresh_token:
                self.log("🔄 تجديد token...")
                try:
                    creds.refresh(Request())
                    self.log("✅ تم تجديد token بنجاح")
                    with open(token_file, 'w') as token:
                        token.write(creds.to_json())
                    self.log("💾 تم حفظ token المجدد")
                except Exception as e:
                    self.log(f"❌ فشل تجديد token: {e}", "ERROR")
                    creds = None
            
            if not creds:
                self.log("🌐 فتح المتصفح للمصادقة...")
                if not os.path.exists(self.config['credentials_file']):
                    raise FileNotFoundError(f"ملف {self.config['credentials_file']} غير موجود!")
                
                flow = InstalledAppFlow.from_client_secrets_file(
                    self.config['credentials_file'], 
                    SCOPES
                )
                creds = flow.run_local_server(port=8080)
                self.log("✅ تمت المصادقة بنجاح")
                
                with open(token_file, 'w') as token:
                    token.write(creds.to_json())
                self.log(f"💾 تم حفظ token في: {token_file}")
            
            self.service = build('blogger', 'v3', credentials=creds)
            self.log("✅ تم بناء Blogger Service بنجاح")
            return True
            
        except Exception as e:
            self.last_error = str(e)
            self.log(f"❌ خطأ في المصادقة: {e}", "ERROR")
            self.log(traceback.format_exc(), "DEBUG")
            return False
    
    def test_connection(self):
        """اختبار الاتصال بالمدونة"""
        if not self.service:
            return False, "الخدمة غير مهيأة"
        
        try:
            blog_id = self.config['blog_id']
            self.log(f"🔍 اختبار الاتصال بالمدونة: {blog_id}")
            
            blog = self.service.blogs().get(blogId=blog_id).execute()
            
            blog_name = blog.get('name', 'غير معروف')
            blog_url = blog.get('url', 'غير معروف')
            pages_count = blog.get('pages', {}).get('totalItems', 0)
            
            self.log(f"✅ المدونة: {blog_name}")
            self.log(f"🌐 الرابط: {blog_url}")
            self.log(f"📄 عدد الصفحات: {pages_count}")
            
            return True, {
                'name': blog_name,
                'url': blog_url,
                'pages': pages_count
            }
            
        except Exception as e:
            self.last_error = str(e)
            self.log(f"❌ فشل الاتصال: {e}", "ERROR")
            return False, str(e)
    
    def publish_page(self, title, content):
        """نشر صفحة (Page) على Blogger"""
        if not self.service:
            return None, "الخدمة غير مهيأة"
        
        try:
            blog_id = self.config['blog_id']
            self.log(f"📤 جاري نشر صفحة على {self.config['name']}: {title}")
            
            page_body = {
                'title': title,
                'content': content
            }
            
            page = self.service.pages().insert(
                blogId=blog_id, 
                body=page_body
            ).execute()
            
            url = page.get('url')
            page_id = page.get('id')
            
            self.log(f"✅ تم نشر الصفحة بنجاح!")
            self.log(f"🆔 Page ID: {page_id}")
            self.log(f"🌐 URL: {url}")
            
            return url, None
            
        except Exception as e:
            self.last_error = str(e)
            self.log(f"❌ فشل نشر الصفحة: {e}", "ERROR")
            self.log(traceback.format_exc(), "DEBUG")
            return None, str(e)
    
    def get_debug_info(self):
        """الحصول على معلومات التصحيح"""
        return "\n".join(self.debug_log[-50:])

    def get_current_blog_key(self):
        """الحصول على مفتاح المدونة المختارة حاليًا"""
        idx = self.blog_combo.current()
        if idx >= 0 and idx < len(self.blog_keys):
            return self.blog_keys[idx]
        return None
    
    def get_current_manager(self):
        """الحصول على مدير المدونة المختارة حاليًا"""
        key = self.get_current_blog_key()
        if key and key in blogger_managers:
            return blogger_managers[key]
        return None
    
    def update_blog_info(self):
        """تحديث معلومات المدونة المختارة"""
        key = self.get_current_blog_key()
        if key:
            config = BLOGGER_CONFIGS[key]
            manager = blogger_managers[key]
            status = "✅ متصل" if manager.service else "⚪ غير متصل"
            self.blog_info_label.config(
                text=f"Blog ID: {config['blog_id'][:15]}... | {status}",
                foreground="#10B981" if manager.service else "#F59E0B"
            )
        else:
            self.blog_info_label.config(text="")
    
    def on_blog_change(self, event=None):
        """عند تغيير المدونة المختارة"""
        self.update_blog_info()
        key = self.get_current_blog_key()
        if key:
            name = BLOGGER_CONFIGS[key]['name']
            self.status_label.config(text=f"🌐 المدونة المختارة: {name}", foreground="#10B981")

    def get_extra_info(self):
        return {
            'league': self.league_entry.get().strip() or 'دوري كرة القدم',
            'channel': self.channel_entry.get().strip() or 'قناة رياضية',
            'commentator': self.commentator_entry.get().strip() or 'غير معروف',
            'stadium': self.stadium_entry.get().strip() or 'ملعب غير معروف'
        }
    
    def analyze_matches(self):
        text = self.matches_input.get("1.0", tk.END)
        self.matches = parse_matches(text)
        
        if not self.matches:
            messagebox.showwarning("تنبيه", "❌ لم يتم العثور على مباريات صحيحة!")
            return
        
        self.matches_listbox.delete(0, tk.END)
        for i, match in enumerate(self.matches):
            title = f"{match['team1']} × {match['team2']}" if match['team2'] else match['team1']
            urls_count = len(match['urls'])
            display_text = f"⏰ {match['time']} | {title} | 🔗 {urls_count} رابط"
            self.matches_listbox.insert(tk.END, display_text)
        
        self.status_label.config(text=f"✅ تم تحليل {len(self.matches)} مباراة", foreground="#10B981")
        self.selected_match = None
        self.generated_pages = []
        self.match_info_label.config(text="🎯 اختر مباراة (أو عدة مباريات) من القائمة")
    
    def select_all_matches(self):
        if not self.matches:
            messagebox.showwarning("تنبيه", "⚠️ لا توجد مباريات!")
            return
        self.matches_listbox.select_set(0, tk.END)
        count = len(self.matches)
        self.match_info_label.config(text=f"✅ تم تحديد {count} مباراة")
        self.status_label.config(text=f"✅ تم تحديد {count} مباراة", foreground="#10B981")
    
    def deselect_all_matches(self):
        self.matches_listbox.select_clear(0, tk.END)
        self.match_info_label.config(text="🎯 اختر مباراة (أو عدة مباريات) من القائمة")
        self.status_label.config(text="🗑️ تم إلغاء التحديد", foreground="#F59E0B")
    
    def on_match_select(self, event):
        selection = self.matches_listbox.curselection()
        if not selection:
            return
        
        if len(selection) == 1:
            index = selection[0]
            self.selected_match = self.matches[index]
            match = self.selected_match
            title = f"{match['team1']} × {match['team2']}" if match['team2'] else match['team1']
            info = f"✅ {title} | ⏰ {match['time']} | 🔗 {len(match['urls'])} رابط"
            self.match_info_label.config(text=info)
            self.status_label.config(text=f"✅ تم اختيار: {title}", foreground="#10B981")
        else:
            self.selected_match = self.matches[selection[0]]
            info = f"🎯 تم تحديد {len(selection)} مباراة (Ctrl/Shift للاختيار)"
            self.match_info_label.config(text=info)
            self.status_label.config(text=f"🎯 تم تحديد {len(selection)} مباراة", foreground="#10B981")
    
    def get_selected_matches(self):
        selection = self.matches_listbox.curselection()
        if not selection:
            return []
        return [self.matches[i] for i in selection]
    
    def check_selection(self):
        if not self.selected_match:
            messagebox.showwarning("تنبيه", "⚠️ الرجاء اختيار مباراة أولاً!")
            return False
        return True
    
    def generate_card(self):
        if not self.check_selection(): return
        html = generate_match_card(self.selected_match, self.get_extra_info())
        self.output_text.delete("1.0", tk.END)
        self.output_text.insert(tk.END, html)
        self.status_label.config(text="📋 تم توليد بطاقة Blogger", foreground="#10B981")
    
    def generate_page(self):
        if not self.check_selection(): return
        html = generate_full_page(self.selected_match, self.get_extra_info())
        self.output_text.delete("1.0", tk.END)
        self.output_text.insert(tk.END, html)
        self.status_label.config(text="🌐 تم توليد صفحة كاملة", foreground="#10B981")
    
    def generate_mochril(self):
        if not self.check_selection(): return
        html = generate_mochril_page(self.selected_match)
        self.output_text.delete("1.0", tk.END)
        self.output_text.insert(tk.END, html)
        self.status_label.config(text="🔴 تم توليد صفحة موشريل", foreground="#10B981")
    
    def generate_all_cards(self):
        selected = self.get_selected_matches()
        if not selected:
            messagebox.showwarning("تنبيه", "⚠️ الرجاء اختيار مباراة واحدة على الأقل!")
            return
        
        self.output_text.delete("1.0", tk.END)
        extra = self.get_extra_info()
        all_cards_html = ""
        
        for match in selected:
            card_html = generate_match_card(match, extra)
            all_cards_html += card_html + "\n"
        
        self.output_text.insert(tk.END, all_cards_html)
        self.generated_pages = []
        for match in selected:
            title = f"{match['team1']} × {match['team2']}" if match['team2'] else match['team1']
            self.generated_pages.append({
                'title': title,
                'content': generate_match_card(match, extra),
                'match': match
            })
        
        self.status_label.config(text=f"📑 تم توليد {len(selected)} بطاقة (كود واحد متصل)", foreground="#10B981")

    def authenticate_current_blog(self):
        """مصادقة المدونة المختارة حاليًا"""
        manager = self.get_current_manager()
        if not manager:
            messagebox.showwarning("تنبيه", "⚠️ لم يتم اختيار مدونة!")
            return
        
        blog_name = manager.config['name']
        self.status_label.config(text=f"🔐 جاري المصادقة: {blog_name}...", foreground="#F59E0B")
        self.update()
        
        issues = manager.check_prerequisites()
        if issues:
            error_msg = f"⚠️ مشاكل في المتطلبات لـ {blog_name}:\n\n" + "\n".join(issues)
            error_msg += "\n\nهل تريد المتابعة على أي حال؟"
            if not messagebox.askyesno("تحذير", error_msg):
                return
        
        success = manager.authenticate()
        
        if success:
            self.update_blog_info()
            self.status_label.config(text=f"✅ تمت المصادقة: {blog_name}", foreground="#10B981")
            messagebox.showinfo("نجاح", f"✅ تمت المصادقة مع {blog_name} بنجاح!")
        else:
            self.status_label.config(text=f"❌ فشلت المصادقة: {blog_name}", foreground="#EF4444")
            error_detail = manager.last_error or "خطأ غير معروف"
            messagebox.showerror("خطأ", f"❌ فشلت المصادقة!\n\nالمدونة: {blog_name}\nالتفاصيل:\n{error_detail}")

        if not manager:
            messagebox.showwarning("تنبيه", "⚠️ لم يتم اختيار مدونة!")
            return

            return
        
        blog_name = manager.config['name']
        title = f"{self.selected_match['team1']} × {self.selected_match['team2']}" if self.selected_match['team2'] else self.selected_match['team1']

            url, error = manager.publish_page(title, code)

                self.after(0, lambda: self.status_label.config(text=f"✅ تم النشر على {blog_name}!", foreground="#10B981"))

        threading.Thread(target=do_publish, daemon=True).start()
    
    def publish_all_selected(self):
        """نشر جميع المباريات المختارة على المدونة المختارة"""
        selected = self.get_selected_matches()
        
        if not selected:
            messagebox.showwarning("تنبيه", "⚠️ الرجاء اختيار مباراة واحدة على الأقل!")
            return
        
        manager = self.get_current_manager()
        if not manager:
            messagebox.showwarning("تنبيه", "⚠️ لم يتم اختيار مدونة!")
            return
        
        if not manager.service:
            blog_name = manager.config['name']
            messagebox.showwarning("تنبيه", f"⚠️ يجب المصادقة أولاً!\nالمدونة: {blog_name}\nاضغط '🔐 مصادقة'")
            return
        
        blog_name = manager.config['name']
        confirm = messagebox.askyesno(
            "تأكيد النشر الجماعي",
            f"📝 المدونة المستهدفة: {blog_name}\n\n"
            f"سيتم نشر {len(selected)} مباراة كصفحات مستقلة.\n\n"
            "كل مباراة ستكون صفحة منفصلة.\n\n"
            "هل تريد المتابعة؟"
        )
        
        if not confirm:
            return
        
        self.status_label.config(text=f"⏳ جاري نشر {len(selected)} مباراة على {blog_name}...", foreground="#F59E0B")
        self.update()
        
        def do_publish_all():
            extra = self.get_extra_info()
            results = []
            
            for i, match in enumerate(selected, 1):
                title = f"{match['team1']} × {match['team2']}" if match['team2'] else match['team1']
                html = generate_full_page(match, extra)
                
                self.after(0, lambda t=title, n=i: self.status_label.config(
                    text=f"⏳ [{n}/{len(selected)}] {t} → {blog_name}", 
                    foreground="#F59E0B"
                ))
                
                url, error = manager.publish_page(title, html)
                results.append({
                    'title': title,
                    'url': url,
                    'error': error
                })
                
                time.sleep(1)
            
            success_count = sum(1 for r in results if r['url'])
            fail_count = len(results) - success_count
            
            report = f"📝 المدونة: {blog_name}\n"
            report += f"✅ تم نشر {success_count} مباراة بنجاح\n"
            report += f"❌ فشل نشر {fail_count} مباراة\n\n"
            report += "📋 التفاصيل:\n" + "="*50 + "\n"
            
            for r in results:
                if r['url']:
                    report += f"✅ {r['title']}\n   🌐 {r['url']}\n\n"
                else:
                    report += f"❌ {r['title']}\n   ⚠️ {r['error']}\n\n"
            
            self.after(0, lambda: self.status_label.config(
                text=f"✅ {success_count}/{len(results)} على {blog_name}", 
                foreground="#10B981"
            ))
            self.after(0, lambda: messagebox.showinfo("نتائج النشر الجماعي", report))
        
        threading.Thread(target=do_publish_all, daemon=True).start()
