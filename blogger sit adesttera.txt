                    self.key = f.read()

    def encrypt_data(self, data):
        if not self.cipher:
            return json.dumps(data)
        try:
            encrypted = self.cipher.encrypt(json.dumps(data).encode())
            return base64.b64encode(encrypted).decode()
        except:
            return json.dumps(data)
    
    def decrypt_data(self, encrypted_str):
        if not self.cipher:
            return json.loads(encrypted_str)
        try:
            encrypted = base64.b64decode(encrypted_str.encode())
            decrypted = self.cipher.decrypt(encrypted)
            return json.loads(decrypted.decode())
        except:
            return json.loads(encrypted_str)

class ResidentialProxyManager:
    def __init__(self):
        self.proxies = []
        self.index = 0
        self.lock = threading.Lock()
        self.fallback_to_free = False

    def load_proxies(self, proxy_text):
        self.proxies = []
        if not proxy_text.strip():
            self.fallback_to_free = True
            return
            
        lines = proxy_text.strip().split('\n')
        for line in lines:
            line = line.strip()
            if not line: continue
            
            # Format: http://user:pass@ip:port
            if '://' in line:
                self.proxies.append(line)
            # Format: ip:port:user:pass
            elif line.count(':') == 3:
                parts = line.split(':')
                self.proxies.append(f"http://{parts[2]}:{parts[3]}@{parts[0]}:{parts[1]}")
            # Format: ip:port
            elif line.count(':') == 1:
                self.proxies.append(f"http://{line}")
            else:
                self.proxies.append(f"http://{line}")
                
        log(f"[✅] تم تحميل {len(self.proxies)} Residential Proxy")
        self.fallback_to_free = len(self.proxies) == 0

    def get_next_proxy(self):
        with self.lock:
            if not self.proxies: return None
            proxy = self.proxies[self.index % len(self.proxies)]
            self.index += 1
            return proxy

@app.route('/stop', methods=['POST'])
def stop_bot():
    bot_state['stop_requested'] = True
    return jsonify({'status': 'stopping'})

@app.route('/stream')
def stream():
    def generate():
        while True:
            try:
                msg = log_queue.get(timeout=1.0)
                yield f"data: {msg}\n\n"
                if msg == "[DONE]": break
            except queue.Empty:
                if not bot_state['running'] and log_queue.empty():
                    yield f"data: [DONE]\n\n"
                    break
                continue
    response = Response(generate(), mimetype='text/event-stream')
    response.headers['Cache-Control'] = 'no-cache'
    return response
